/** * JC MECHANICAL - LIVE KPI DASHBOARD WITH DFO TRACKING * Google Apps Script Backend * * This script: * 1. Reads Housecall Pro transaction data from Google Sheets * 2. Calculates weekly revenue totals * 3. Compares actual vs budget * 4. Tracks DFO (Diagnostic Fee Only) percentage * 5. Serves data as JSON to the HTML dashboard */ // ============================================================================ // CONFIGURATION // ============================================================================ const CONFIG = { // Your 2026 Annual Budget ANNUAL_BUDGET: 1300000, // Monthly Revenue Distribution Percentages MONTHLY_PERCENTAGES: { 'Jan': 7.47, 'Feb': 5.84, 'Mar': 4.43, 'Apr': 9.50, 'May': 10.49, 'Jun': 16.07, 'Jul': 9.21, 'Aug': 11.00, 'Sep': 9.17, 'Oct': 4.96, 'Nov': 6.46, 'Dec': 5.40 }, // Sheet names in your Google Sheet SHEETS: { RAW_DATA: 'Housecall Pro Data', BUDGET_TARGETS: 'Budget Targets', WEEKLY_ACTUALS: 'Weekly Actuals', DASHBOARD_FEED: 'Dashboard Data' } }; // ============================================================================ // WEB APP ENDPOINT (for live dashboard) // ============================================================================ function doGet(e) { try { const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'dashboard'; let data; switch(action) { case 'dashboard': data = getDashboardData(); break; case 'weekly': data = getWeeklyData(); break; case 'monthly': data = getMonthlyData(); break; default: data = getDashboardData(); } return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { Logger.log('Error in doGet: ' + error.toString()); return ContentService .createTextOutput(JSON.stringify({ error: error.toString(), message: 'Dashboard API Error' })) .setMimeType(ContentService.MimeType.JSON); } } // ============================================================================ // MAIN DASHBOARD DATA GENERATOR // ============================================================================ function getDashboardData() { const ss = SpreadsheetApp.getActiveSpreadsheet(); // Get or create sheets const rawDataSheet = getOrCreateSheet(ss, CONFIG.SHEETS.RAW_DATA); const budgetSheet = getOrCreateSheet(ss, CONFIG.SHEETS.BUDGET_TARGETS); // Calculate budget targets if not already done if (budgetSheet.getLastRow() < 2) { calculateBudgetTargets(budgetSheet); } // Get weekly actuals from raw data const weeklyActuals = calculateWeeklyActuals(rawDataSheet); // Get budget targets const weeklyBudgets = getWeeklyBudgets(budgetSheet); // Calculate service metrics (DFO tracking) const serviceMetrics = calculateServiceMetrics(rawDataSheet); // Compare and generate dashboard data const dashboardData = { timestamp: new Date().toISOString(), annual: { budget: CONFIG.ANNUAL_BUDGET, actualYTD: 0, percentComplete: 0 }, months: [], weeks: [], summary: { bestWeek: null, worstWeek: null, avgWeekly: 0, onTrack: false }, serviceMetrics: serviceMetrics }; // Process each month const months = Object.keys(CONFIG.MONTHLY_PERCENTAGES); let totalActual = 0; months.forEach(month => { const monthData = { name: month, percent: CONFIG.MONTHLY_PERCENTAGES[month], budget: CONFIG.ANNUAL_BUDGET * (CONFIG.MONTHLY_PERCENTAGES[month] / 100), actual: 0, variance: 0, variancePercent: 0, weeks: [] }; // Add weekly data for this month for (let week = 1; week <= 4; week++) { const weekKey = `${month}_W${week}`; const actual = weeklyActuals[weekKey] || 0; const budget = weeklyBudgets[weekKey] || 0; monthData.actual += actual; totalActual += actual; const weekData = { week: week, weekKey: weekKey, actual: actual, budget: budget, variance: actual - budget, variancePercent: budget > 0 ? ((actual / budget) * 100) : 0 }; monthData.weeks.push(weekData); dashboardData.weeks.push(weekData); } monthData.variance = monthData.actual - monthData.budget; monthData.variancePercent = monthData.budget > 0 ? ((monthData.actual / monthData.budget) * 100) : 0; dashboardData.months.push(monthData); }); // Calculate annual summary dashboardData.annual.actualYTD = totalActual; dashboardData.annual.percentComplete = (totalActual / CONFIG.ANNUAL_BUDGET) * 100; // Find best and worst weeks const weeksWithData = dashboardData.weeks.filter(w => w.actual > 0); if (weeksWithData.length > 0) { dashboardData.summary.bestWeek = weeksWithData.reduce((max, w) => w.actual > max.actual ? w : max ); dashboardData.summary.worstWeek = weeksWithData.reduce((min, w) => w.actual < min.actual ? w : min ); dashboardData.summary.avgWeekly = totalActual / weeksWithData.length; } return dashboardData; } // ============================================================================ // SERVICE CALL METRICS - DFO TRACKING // ============================================================================ function calculateServiceMetrics(sheet) { if (sheet.getLastRow() < 2) { return { totalServiceCalls: 0, diagnosticOnlyCalls: 0, diagnosticPercentage: 0, isGood: true }; } const data = sheet.getDataRange().getValues(); let totalServiceCalls = 0; let dfoCount = 0; // Skip header row, start at row 1 for (let i = 1; i < data.length; i++) { const tags = String(data[i][3]).toLowerCase(); // Column D: Tags // Count all service calls (anything with "service" in the tag) if (tags.includes('service')) { totalServiceCalls++; // Count DFO calls specifically if (tags.includes('dfo')) { dfoCount++; } } } const dfoPercentage = totalServiceCalls > 0 ? (dfoCount / totalServiceCalls) * 100 : 0; return { totalServiceCalls: totalServiceCalls, diagnosticOnlyCalls: dfoCount, diagnosticPercentage: dfoPercentage, isGood: dfoPercentage < 5 // Green if under 5% }; } // ============================================================================ // BUDGET CALCULATIONS // ============================================================================ function calculateBudgetTargets(sheet) { // Clear existing data sheet.clear(); // Headers sheet.getRange('A1:E1').setValues([[ 'Month', 'Percentage', 'Monthly Budget', 'Weekly Budget', 'Week Key' ]]).setFontWeight('bold'); let row = 2; Object.keys(CONFIG.MONTHLY_PERCENTAGES).forEach(month => { const percent = CONFIG.MONTHLY_PERCENTAGES[month]; const monthlyBudget = CONFIG.ANNUAL_BUDGET * (percent / 100); const weeklyBudget = monthlyBudget / 4; for (let week = 1; week <= 4; week++) { sheet.getRange(row, 1, 1, 5).setValues([[ month, percent / 100, monthlyBudget, weeklyBudget, `${month}_W${week}` ]]); row++; } }); // Format sheet.getRange(2, 2, row - 2, 1).setNumberFormat('0.00%'); sheet.getRange(2, 3, row - 2, 2).setNumberFormat('$#,##0.00'); sheet.autoResizeColumns(1, 5); } function getWeeklyBudgets(sheet) { const data = sheet.getDataRange().getValues(); const budgets = {}; for (let i = 1; i < data.length; i++) { const weekKey = data[i][4]; // Column E: Week Key const weeklyBudget = data[i][3]; // Column D: Weekly Budget budgets[weekKey] = weeklyBudget; } return budgets; } // ============================================================================ // ACTUAL DATA CALCULATIONS // ============================================================================ function calculateWeeklyActuals(sheet) { const weeklyTotals = {}; // Check if sheet has data if (sheet.getLastRow() < 2) { return weeklyTotals; } const data = sheet.getDataRange().getValues(); // Columns: Date | Description | Amount | Category for (let i = 1; i < data.length; i++) { const date = new Date(data[i][0]); const amount = parseFloat(data[i][2]) || 0; // Skip if not a valid date or amount if (isNaN(date.getTime()) || amount <= 0) continue; // Determine month and week const monthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec']; const month = monthNames[date.getMonth()]; const weekOfMonth = Math.min(Math.ceil(date.getDate() / 7), 4); const weekKey = `${month}_W${weekOfMonth}`; // Initialize if not exists if (!weeklyTotals[weekKey]) { weeklyTotals[weekKey] = 0; } // Add to weekly total weeklyTotals[weekKey] += amount; } return weeklyTotals; } // ============================================================================ // MONTHLY DATA (for charts) // ============================================================================ function getMonthlyData() { const dashboardData = getDashboardData(); return { labels: dashboardData.months.map(m => m.name), budgets: dashboardData.months.map(m => m.budget), actuals: dashboardData.months.map(m => m.actual), percentages: dashboardData.months.map(m => m.percent) }; } // ============================================================================ // WEEKLY DATA (for detailed views) // ============================================================================ function getWeeklyData() { const dashboardData = getDashboardData(); return { weeks: dashboardData.weeks }; } // ============================================================================ // UTILITY FUNCTIONS // ============================================================================ function getOrCreateSheet(spreadsheet, sheetName) { let sheet = spreadsheet.getSheetByName(sheetName); if (!sheet) { sheet = spreadsheet.insertSheet(sheetName); } return sheet; } // ============================================================================ // HOURLY UPDATE TRIGGER (Optional - set up via Triggers menu) // ============================================================================ function setupHourlyTrigger() { // Delete existing triggers const triggers = ScriptApp.getProjectTriggers(); triggers.forEach(trigger => ScriptApp.deleteTrigger(trigger)); // Create new hourly trigger ScriptApp.newTrigger('refreshDashboardData') .timeBased() .everyHours(1) .create(); } function refreshDashboardData() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const feedSheet = getOrCreateSheet(ss, CONFIG.SHEETS.DASHBOARD_FEED); // Get latest dashboard data const data = getDashboardData(); // Write to sheet as JSON feedSheet.clear(); feedSheet.getRange('A1').setValue('Last Updated: ' + new Date().toLocaleString()); feedSheet.getRange('A2').setValue(JSON.stringify(data, null, 2)); Logger.log('Dashboard data refreshed at ' + new Date()); }
